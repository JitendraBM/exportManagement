{# Shared goods-row renderer for the invoice sheet and the annexure sheet. Groups consecutive items under a category heading row ("FIRST LINE OF NAME - HSNC xxx"), like the original Excel documents: - mode 'index' : every item line gets a running number (printed in the Marks column), numbering continues across the annexure. - mode 'surface' : items additionally split by their surface finish, and the group's SURFACE (GLOSSY / MATT / ...) prints in the Marks column of the heading row; no line numbers. An item whose name has extra lines below the first prints those lines as its description (the first line lives in the heading); a single-line item prints just its quantities under the heading. #} {% macro group_key(item, mode) -%} {{- (item.product_name or '').split('\n')[0]|trim|upper }}|{{ item.hsn_code or '' }}{% if mode == 'surface' %}|{{ (item.surface or '')|trim|upper }}{% endif -%} {%- endmacro %} {% macro rows(items, mode, sr_start=0) %} {% set gs = namespace(sr=sr_start, key=none, first=true) %} {% for item in items %} {% set key = group_key(item, mode) %} {% set is_new_group = key != gs.key %} {# A group with exactly one item and a single-line name doesn't need its own heading row above a near-empty numbered row - that split only earns its keep when a heading actually has multiple variant lines under it (surface finishes, or a multi-line product name). Detected by peeking at the next item, since group membership can't be known until the group closes. #} {% set next_key = group_key(items[loop.index], mode) if loop.index < items|length else none %} {% set is_solo = is_new_group and next_key != key and '\n' not in (item.product_name or '') %} {% if is_new_group %} {% set gs.key = key %} {% if not gs.first %} {% endif %} {% if not is_solo %} {% if mode == 'surface' %}{{ (item.surface or '')|upper }}{% endif %} {{ (item.product_name or '').split('\n')[0]|upper }}{% if item.hsn_code %} - HSNC {{ item.hsn_code }}{% endif %} {% endif %} {% endif %} {% set gs.first = false %} {% set gs.sr = gs.sr + 1 %} {% set rest = (item.product_name or '').split('\n')[1:] | join('\n') %} {%- if mode == 'index' %}{{ gs.sr }}{% elif is_solo %}{{ (item.surface or '')|upper }}{% endif -%} {%- if is_solo -%} {{ (item.product_name or '').split('\n')[0]|upper }}{% if item.hsn_code %} - HSNC {{ item.hsn_code }}{% endif %} {%- else -%} {%- for line in rest.split('\n') -%} {{ line|upper }}{% if not loop.last %}
{% endif %} {%- endfor -%} {%- endif -%} {{ '{:,.2f}'.format(item.pallets) if item.pallets else '' }} {% if item.quantity_boxes %}{{ '{:,.0f}'.format(item.quantity_boxes) }} {{ item.quantity_unit }}{% endif %} {{ '{:,.2f}'.format(item.quantity_value or 0) }} ${{ '{:,.2f}'.format(item.price_usd or 0) }} {{ item.unit }} {% if item.total_usd %}$ {{ '{:,.2f}'.format(item.total_usd) }}{% else %}$ -{% endif %} {% endfor %} {% endmacro %}
