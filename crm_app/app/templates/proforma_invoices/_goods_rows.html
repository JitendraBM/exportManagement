{# Shared goods-row renderer for the invoice sheet and the annexure sheet. Groups consecutive items under a category heading row (first line of the name, HSN code in its own column), like the original Excel documents. Every item line gets a running number in the Sr No column, numbering continues across the annexure, in both display modes: - mode 'index' : items are grouped by name/HSN only. - mode 'surface' : items are additionally split by their surface finish, and the group's SURFACE (GLOSSY / MATT / ...) prints in the Sr No column of the heading row - so a heading row always prints in this mode, even for a lone item, to keep the surface visible without covering up its line number below. An item whose name has extra lines below the first prints those lines as its description (the first line lives in the heading); a single-line item prints just its quantities under the heading. #} {% macro group_key(item, mode) -%} {{- (item.product_name or '').split('\n')[0]|trim|upper }}|{{ item.hsn_code or '' }}{% if mode == 'surface' %}|{{ (item.surface or '')|trim|upper }}{% endif -%} {%- endmacro %} {% macro rows(items, mode, sr_start=0, currency_prefix='$') %} {% set gs = namespace(sr=sr_start, key=none, first=true) %} {% for item in items %} {% set key = group_key(item, mode) %} {% set is_new_group = key != gs.key %} {# A group with exactly one item and a single-line name doesn't need its own heading row above a near-empty numbered row - that split only earns its keep when a heading actually has multiple variant lines under it (surface finishes, or a multi-line product name). Detected by peeking at the next item, since group membership can't be known until the group closes. #} {% set next_key = group_key(items[loop.index], mode) if loop.index < items|length else none %} {% set is_solo = is_new_group and next_key != key and '\n' not in (item.product_name or '') %} {# In surface mode the heading row is the only place the surface name prints, so it always prints - even for a solo item - rather than borrowing the item row's own Sr No cell (which would blank out its number). In index mode a solo item still skips the heading, inlining its name into the numbered row as before. #} {% set inline_into_row = is_solo and mode != 'surface' %} {% if is_new_group %} {% set gs.key = key %} {% if not gs.first %} {% endif %} {% if not inline_into_row %} {% if mode == 'surface' %}{{ (item.surface or '')|upper }}{% endif %} {{ (item.product_name or '').split('\n')[0]|upper }} {{ item.hsn_code or '' }} {% endif %} {% endif %} {% set gs.first = false %} {% set gs.sr = gs.sr + 1 %} {% set rest = (item.product_name or '').split('\n')[1:] | join('\n') %} {{ gs.sr }} {%- if inline_into_row -%} {{ (item.product_name or '').split('\n')[0]|upper }} {%- else -%} {%- for line in rest.split('\n') -%} {{ line|upper }}{% if not loop.last %}
{% endif %} {%- endfor -%} {%- endif -%} {% if inline_into_row %}{{ item.hsn_code or '' }}{% endif %} {{ '{:,.2f}'.format(item.pallets) if item.pallets else '' }} {% if item.quantity_boxes %}{{ '{:,.0f}'.format(item.quantity_boxes) }} {{ item.quantity_unit }}{% endif %} {{ '{:,.2f}'.format(item.quantity_value or 0) }} {{ item.unit }} {{ currency_prefix }}{{ '{:,.2f}'.format(item.price_usd or 0) }} {{ item.unit }} {{ currency_prefix }} {{ '{:,.2f}'.format(item.total_usd or 0) }} {% endfor %} {% endmacro %}
