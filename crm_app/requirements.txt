# Core web framework
Flask==3.1.3

# Used by Flask for templating/security (installed automatically with Flask,
# pinned explicitly here so `pip install -r requirements.txt` is reproducible)
Jinja2==3.1.6
Werkzeug==3.1.7

# Used by CurrencyConversionService to call a live exchange-rate API
requests==2.33.1

# Used to load configuration (SECRET_KEY, DB path, etc.) from a .env file
python-dotenv==1.2.2

# Used to read uploaded .xlsx files (Booking Detail's "Upload from Excel" -
# app/xlsx.py writes .xlsx by hand to avoid this dependency, but parsing an
# arbitrary user-uploaded sheet is worth a real library)
openpyxl==3.1.5

# Used by the Packing Planning label sheet to encode one QR per physical
# pallet (pure Python, no dependencies of its own, emits SVG directly). The
# hand-rolled route app/docx.py and app/xlsx.py take was rejected here: a QR's
# Reed-Solomon and mask selection have to be exactly right or the labels don't
# scan, and that only shows up once they are stuck on pallets.
segno==1.6.6

# NOTE: The database layer uses Python's built-in `sqlite3` module, so no
# separate database driver is required. See README.md -> "Swapping the
# database" for how to move to PostgreSQL/MySQL later without touching
# services or routes (that's the point of the Repository pattern used here).

# Test-only dependency (the regression suite under tests/). Not needed to run
# the app itself. Run the suite with:  python -m pytest
pytest==9.1.1
